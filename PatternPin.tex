\documentclass{article}

% Define the grid size
\usepackage{tikz}
\usetikzlibrary{shapes.geometric}

% Define the shape of the pin
\newcommand{\pinshape}{
    [
        shape=regular polygon,
        regular polygon sides=6,
        minimum size=2cm,
        draw=black,
        fill=white
    ]
}

% Define the symbols used in the pattern
\newcommand{\symbols}{
    \{,+,*,#,$,%,^, &, |, (, ), ~, !, @, #, $, %, &, |, +, -, =, <, >, /, :, ;, ,, ., ?, @, \#, \$
}

% Function to generate a hash from a shape
\newcommand{\generatehash}{
    \begin{tikzpicture}
        \node[pinshape] (pin) {};
        \coordinate (center) at (pin.center);
        \foreach \i in {1,...,6} {
            \pgfmathsetmacro{\angle}{360/\number\numexpr6+\number\numexpr\i-1*\number\numexpr6+\number\numexpr\i-1}
            \pgfmathsetmacro{\x}{cos(\angle)}
            \pgfmathsetmacro{\y}{sin(\angle)}
            \coordinate (point\i) at ($(center) + (\x,\y)$);
            \node[anchor=center] at (point\i) {$\alpha_{\i}$};
        }
    \end{tikzpicture}
}
